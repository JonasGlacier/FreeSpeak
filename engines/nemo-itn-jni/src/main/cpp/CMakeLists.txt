cmake_minimum_required(VERSION 3.22.1)
project(freespeak_itn LANGUAGES CXX)
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
foreach(feature BIN SCRIPT COMPACT COMPRESS CONST FSTS GRM LINEAR LOOKAHEAD MPDT NGRAM PDT SPECIAL)
    set(HAVE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(HAVE_FAR ON CACHE BOOL "" FORCE)
FetchContent_Declare(openfst
    URL https://github.com/csukuangfj/openfst/archive/refs/tags/v1.8.5-2026-11-21.tar.gz
    URL_HASH SHA256=9ea090fd9b17cf4edfbc47ab49f0645d1b70eb48cd21047c1897509c379a9ef1)
FetchContent_MakeAvailable(openfst)
foreach(target fst fstfar)
    set_target_properties(${target} PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
endforeach()
add_library(freespeak_itn SHARED nemo_itn_jni.cpp)
target_compile_features(freespeak_itn PRIVATE cxx_std_17)
target_compile_options(freespeak_itn PRIVATE -O3 -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_options(freespeak_itn PRIVATE -Wl,--exclude-libs,ALL -Wl,--no-undefined)
target_link_libraries(freespeak_itn PRIVATE fstfar fst)

# Explicit opt-in target: synthetic runtime checks, never packaged in the APK.
add_executable(itn_runtime_test EXCLUDE_FROM_ALL runtime_test.cpp)
target_compile_features(itn_runtime_test PRIVATE cxx_std_17)
target_link_libraries(itn_runtime_test PRIVATE fstfar fst)
